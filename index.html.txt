            const totalConsumed = consumption.reduce((s,c)=>s+c.amount,0);
            const remaining = totalDeposited - totalConsumed;
            const totalSpent = deposits.reduce((s,d)=>s+d.cost,0);

        async function renderDashboardCharts(deposits, consumption) {
            const assets = await getDocs(COLLECTIONS.assets);
            // Daily chart
            const days = [], dailyData = [];
            const now = new Date();
            for (let d=13; d>=0; d--) {
                const date = new Date(now); date.setDate(date.getDate()-d);
                const dateStr = date.toISOString().split('T')[0];
                days.push(dateStr.slice(5));
                const total = consumption.filter(c=>c.date===dateStr).reduce((s,c)=>s+c.amount,0);
                dailyData.push(Math.round(total*10)/10);
            }
            if (dailyChart) dailyChart.destroy();
            dailyChart = new Chart(document.getElementById('dailyConsumptionChart'), {
                type:'bar', data:{labels:days,datasets:[{label:'Liters',data:dailyData,backgroundColor:'rgba(30,111,217,0.7)',borderRadius:6}]},
                options:{responsive:true,plugins:{legend:{display:false}},scales:{y:{beginAtZero:true,grid:{color:'#f1f5f9'}},x:{grid:{display:false}}}}
            });
            // Pie chart
            const assetMap = {};
            consumption.forEach(c=>{ const a=assets.find(x=>x.id===c.assetId); const name=a?a.name:'Unknown'; assetMap[name]=(assetMap[name]||0)+c.amount; });
            const labels=Object.keys(assetMap), data=Object.values(assetMap);
            if (pieChart) pieChart.destroy();
            if (labels.length===0) {
                pieChart = new Chart(document.getElementById('fuelPieChart'), {type:'doughnut',data:{labels:['No Data'],datasets:[{data:[1],backgroundColor:['#e2e8f0']}]}});
            } else {
                pieChart = new Chart(document.getElementById('fuelPieChart'), {
                    type:'doughnut', data:{labels,datasets:[{data,backgroundColor:['#1e6fd9','#f59e0b','#10b981','#ef4444','#8b5cf6','#ec4899']}]},
                    options:{plugins:{legend:{position:'bottom',labels:{padding:16,usePointStyle:true}}}}
                });
            }
        }

        // ==================== ACCOUNTING ====================
        async function renderAccounting() {
            const [deposits, consumption, assets] = await Promise.all([getDocs(COLLECTIONS.deposits), getDocs(COLLECTIONS.consumption), getDocs(COLLECTIONS.assets)]);
            const totalSpent = deposits.reduce((s,d)=>s+d.cost,0);
            const totalDep = deposits.reduce((s,d)=>s+d.amount,0);
            const avgCost = totalDep>0 ? totalSpent/totalDep : 0;
            const totalCons = consumption.reduce((s,c)=>s+c.amount,0);
            document.getElementById('accountingCards').innerHTML = `
                <div class="summary-card accent-blue"><div class="card-label">💵 Total Spent</div><div class="card-value">$${totalSpent.toFixed(2)}</div></div>
                <div class="summary-card accent-green"><div class="card-label">📊 Avg $/L</div><div class="card-value">$${avgCost.toFixed(3)}</div></div>
                <div class="summary-card accent-orange"><div class="card-label">📉 Est. Cost Consumed</div><div class="card-value">$${(totalCons*avgCost).toFixed(2)}</div></div>
                <div class="summary-card accent-red"><div class="card-label">🔢 Deposits</div><div class="card-value">${deposits.length}</div></div>
            `;
            const assetMap = {};
            consumption.forEach(c=>{ const a=assets.find(x=>x.id===c.assetId); const name=a?a.name:'Unknown'; if(!assetMap[name]) assetMap[name]={consumed:0,type:a?a.type:'unknown'}; assetMap[name].consumed+=c.amount; });
            document.getElementById('accountingTable').innerHTML = Object.keys(assetMap).length===0 ? '<tr><td colspan="5">No data.</td></tr>' : Object.entries(assetMap).map(([name,d])=>`<tr><td>${name}</td><td><span class="badge ${d.type==='car'?'badge-car':'badge-generator'}">${d.type}</span></td><td>${d.consumed.toFixed(1)} L</td><td>$${avgCost.toFixed(3)}</td><td>$${(d.consumed*avgCost).toFixed(2)}</td></tr>`).join('');
        }

        async function saveRecord(type) {
            const getVal = id => document.getElementById(id)?.value;
            switch(type) {
                case 'consumption':{
                    const date=getVal('modDate'), assetId=getVal('modAssetId'), amount=parseFloat(getVal('modAmount'));
                    if(!date||!assetId||isNaN(amount)||amount<=0) return alert('Fill all fields');
                    await addDoc(COLLECTIONS.consumption,{date,assetId,amount,recordedBy:currentUser.email});
                    break;
                }
                case 'allocation':{
                    const date=getVal('modDate'), assetId=getVal('modAssetId'), amount=parseFloat(getVal('modAmount')), purpose=getVal('modPurpose')||'General';
                    if(!date||!assetId||isNaN(amount)||amount<=0) return alert('Fill all fields');
                    await addDoc(COLLECTIONS.allocations,{date,assetId,amount,purpose,allocatedBy:currentUser.email});
                    break;
                }
                case 'deposit':{
                    const date=getVal('modDate'), amount=parseFloat(getVal('modAmount')), cost=parseFloat(getVal('modCost')), supplier=getVal('modSupplier')||'Unknown', notes=getVal('modNotes')||'';
                    if(!date||isNaN(amount)||amount<=0||isNaN(cost)||cost<=0) return alert('Fill all fields');
                    await addDoc(COLLECTIONS.deposits,{date,amount,cost,supplier,notes});
                    break;
                }
                case 'asset':{
                    const name=getVal('modName'), type=getVal('modType'), identifier=getVal('modIdentifier')||'';
                    if(!name||!type) return alert('Fill all fields');
                    await addDoc(COLLECTIONS.assets,{name,type,identifier});
                    break;
                }
            }
            closeModal();
            refreshAllViews();
        }
