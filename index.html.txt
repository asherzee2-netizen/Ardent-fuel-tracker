            if (user) {
                // Fetch user role from Firestore
                const userDoc = await db.collection('users').doc(user.uid).get();
                if (userDoc.exists) {
                    const userData = userDoc.data();
                    currentUser = {
                        uid: user.uid,
                        email: user.email,
                        role: userData.role || 'operator'
                    };

                errorEl.textContent = 'Please enter email and password.';
                return;

            }
